# Shared hosts where blocking must be URL-level (host/path), not whole-host.
# Format: <host> <regex matched against the URL path>. The matched part becomes
# the urls.txt entry, e.g. sites.google.com/view/coolgames
# (GitHub/npm CDNs such as jsDelivr and unpkg are handled in collector/cdn.py.)
sites.google.com    ^/(?:view/[^/?#]+|a/[^/]+/[^/?#]+|(?!view/|a/|u/|new|s/)[^/]+\.[^/]+/[^/?#]+)
script.google.com   ^/(?:a/macros/[^/]+/s/[^/]+|macros/s/[^/]+)
codehs.com          ^/(?:sandbox|share)/[^/]+/[^/?#]+
